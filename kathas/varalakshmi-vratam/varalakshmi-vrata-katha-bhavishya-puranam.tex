\sect{कथा {[भविष्योत्तरपुराणम्]}}

%source VrataRatnakaramu (Telugu)


\twolineshloka
{कैलासशिखरे रम्ये नानागणनिषेविते}
{मन्दारविटपिप्रीते नानामणिविभूषिते} %१

\twolineshloka
{पाटलाशोकपुन्नागखर्जूरवकुलान्विते}
{कुबेरवरुणेन्द्रादिदिक्पालैश्च समावृते} %२

\twolineshloka
{नारदागस्त्यवाल्मीकिपराशरसमावृते}।
{रत्नपीठे सुखासीनं शङ्करं लोकशङ्करम्} %३

\onelineshloka*
{पप्रच्छ गौरी सन्तुष्टा लोकानुग्रहकाम्यया}

\uvacha{पार्वत्युवाच}

\onelineshloka*
{भगवन् सर्वलोकेश सर्वभूतहिते रत} %४

\onelineshloka*
{यद्रहस्यमिदं पुण्यं तदाचक्ष्व ममानघ}

\uvacha{ईश्वर उवाच}

\onelineshloka
{व्रतानामुत्तमं नाम सर्वसौभाग्यकारणम्} %५

\twolineshloka
{सर्वसम्पत्प्रदं शीघ्रं पुत्रपौत्रप्रवर्धनम्}
{वरलक्ष्मीव्रतं नाम व्रतमस्ति मनोहरम्} %६

\twolineshloka
{शुक्ले श्रावणिके मासे पूर्णिमोपान्त्यभार्गवे}
{यदा तु नारी वर्तेत व्रते तस्याः फलं शृणु} %७

\uvacha{पार्वत्युवाच}

\twolineshloka
{विधिना केन कर्तव्यं तत्र का नाम देवता}
{कया चाऽऽराधिता पूर्वं साऽभूत् सन्तुष्टमानसा} %८

\uvacha{ईश्वर उवाच}

\twolineshloka
{वरलक्ष्मीव्रतं पुण्यं वक्ष्यामि शृणु पार्वति}
{कुण्डिनं नाम नगरं सर्वमण्डनमण्डितम्} %९

\twolineshloka
{हेमप्राकारसंयुक्तं चामीकरगृहोज्ज्वलम्}
{तत्राभूद् ब्राह्मणी काचिन्नाम्ना चारुमती श्रुता} %१०

\twolineshloka
{पतिभक्तिरता साध्वी श्वश्रूश्वशुरयोर्मता}
{कलावती सा विदुषी सततं मञ्जुभाषिणी} %११

\twolineshloka
{तस्याः प्रसन्नचित्ताया लक्ष्मीः स्वप्नगता तदा}
{एहि कल्याणि भद्रं ते वरलक्ष्मीः समागता} %१२

\twolineshloka
{नभोमासे पूर्णिमायां नातिक्रान्ते भृगोर्दिने}
{मत्पूजा तत्र कर्तव्या वरं दास्यामि काङ्क्षितम्} %१३

\twolineshloka
{इत्युक्तां वरलक्ष्मीं सा तुष्टाव परया मुदा}
{नमस्ते सर्वलोकानां जनन्यै पुण्यमूर्तये} %१४

\twolineshloka
{शरण्ये त्रिजगद्वन्द्ये विष्णुवक्षःस्थलालये}
{त्वयाऽवलोकितः सद्यः स धन्यः स गुणान्वितः} %१५

\twolineshloka
{स श्लाघ्यः स कुशलम्बी च स शूरः स च पण्डितः}
{जन्मान्तरसहस्रेषु किं मया सुकृतं कृतम्} %१६

\twolineshloka
{अतस्त्वत्पादयुगलं पश्यामि हरिवल्लभे}
{एवं स्तुता सा कमला प्रादात् तस्यै बहुं वरान्} %१७

\twolineshloka
{ततश्चारुमती साध्वी स्वप्नादुत्थाय सम्भ्रमात्}
{तत्सर्वं कथयामास बन्धूनां पुरतस्तदा} %१८

\twolineshloka
{श्रुत्वा तु बान्धवाः सर्वे साधु साध्विति चाब्रुवन्}
{तथैव करवामेति तदागमनकाङ्क्षिणः} %१९

\twolineshloka
{भाग्योदयेन सम्प्राप्ते वरलक्ष्मीदिने तथा}
{स्त्रियः प्रसन्नहृदया निर्मलाश्चित्रवाससः} %२०

\twolineshloka
{नूतनैस्तण्डुलैः पूर्णे कुम्भे च वटपल्लवैः}
{सायं चारुमतीमुख्याश्चक्रुः पूजां प्रयत्नतः} %२१

\twolineshloka
{पद्मासने पद्मकरे सर्वलोकैकपूजिते}
{नारायणप्रिये देवि सुप्रीता भव सर्वदा} %२२

\twolineshloka
{इत्यादिमन्त्रैः सकलैरुपचारा यथाक्रमम्}
{कृत्वा तु दक्षिणे हस्ते नवसूत्रं दधुः स्त्रियः} %२३

\twolineshloka
{हविष्यं सघृतं चैव वरलक्ष्म्यै न्यवेदयन्}
{गन्धादिभिरलङ्कृत्य सुशीलं वृद्धभूसुरम्} %२४

\twolineshloka
{तस्मै दत्त्वा वायनं च द्वादशापूपसंयुतम्}
{ततो देवीसमीपे तु हविष्यं चक्रुरङ्गनाः} %२५

\twolineshloka
{अथ लक्ष्मीप्रसादेन मुक्तामाणिक्यभूषिताः}
{नूपुराक्रान्तचरणा मणिकाञ्चनभूषणाः} %२६

\twolineshloka
{पुत्रपौत्रैः परिवृता धनधान्यसमृद्धिभिः}
{अन्नदानरता नित्यं बन्धुपोषणतत्पराः} %२७

\twolineshloka
{स्वं स्वं सद्म समाजग्मुर्हस्त्यश्वरथसङ्कुलम्}
{अन्योन्यं कथयामासुः श्रुतं चारुमतीमुखात्} %२८

\twolineshloka
{इदं सत्यमिदं सत्यं नरो भद्राणि पश्यति}
{वयं चारुमतीमुख्या उपलब्धमनोरथाः} %२९

\twolineshloka
{पुण्या चारुमती धन्या भूयो भाग्यवती चिरम्}
{स्वयं यस्यान्महालक्ष्म्या बोधितं हि व्रतोत्तमम्} %३०

\twolineshloka
{इति चारुमतीं साध्वीं तुष्टुवुस्तत्र योषितः}
{वरलक्ष्मीव्रतं नाम तदादि भुवि विश्रुतम्} %३१

\threelineshloka
{एतत्ते सर्वमाख्यातं व्रतानामुत्तमं व्रतम्}
{य इदं शृणुयाद्वापि श्रावयेद्वा समाहितः}
{सिद्ध्यन्ति सर्वकार्याणि वरलक्ष्मीप्रसादतः} %३२
