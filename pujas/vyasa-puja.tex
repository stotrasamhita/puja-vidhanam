\renewcommand{\additionalSankalpa}{
\begin{itemize}
\item उत्तराषाढा-नक्षत्रे धनूराशौ आविर्भू\-तानां श्रीमत्-शङ्कर-विजयेन्द्र-सरस्वती-श्रीपादानां, शतभिषङ्-नक्षत्रे कुम्भ-राशौ आविर्भूतानां श्रीमत्-सत्य-चन्द्रशेखरेन्द्र-सरस्वती-श्रीपादानाम् अस्माकं जगद्गुरूणां दीर्घ-आयुः-आरोग्य-सिद्ध्यर्थं,

\item तैः सङ्कल्पितानां सर्वेषां लोक-क्षेमार्थ-कार्याणां वेद-शास्त्रादि-सम्प्रदाय-पोषण-कार्याणां विविध-क्षेत्र-यात्रायाश्च अविघ्नतया सम्पूर्त्यर्थं

\item कामकोटि-गुरु-परम्परायां कामकोटि-भक्त-जनानाम् अचञ्चल-भावशुद्ध-दृढतर-भक्ति-सिद्ध्यर्थं, परस्पर-ऐकमत्य-सिद्ध्यर्थं

\item भारतीयानां महाजनानां विघ्न-निवृत्ति-पूर्वक-सत्कार्य-प्रवृत्ति-द्वारा ऐहिक-आमुष्मिक-अभ्युदय-प्राप्त्यर्थम्, असत्कार्येभ्यः निवृत्त्यर्थं

\item भारतीयानां सन्ततेः सनातन-सम्प्रदाये श्रद्धा-भक्त्योः अभिवृद्ध्यर्थं

\item सर्वेषां द्विपदां चतुष्पदाम् अन्येषां च प्राणि-वर्गाणाम् आरोग्य-युक्त-सुख-जीवन-अवाप्त्यर्थम्

\item अस्माकं सह-कुटुम्बानां धर्म-अर्थ-काम-मोक्ष-रूप-चतुर्विध-पुरुषार्थ-सिद्ध्यर्थं विवेक-वैराग्य-सिद्ध्यर्थं
\end{itemize}
}

{व्यासाय धाम्ने तपसां संसारायासहारिणे}

॥इति काश्मीरिकेण क्षेमेन्द्रकविना कृतायां भारतमञ्जर्याम् उपसंहारे पठितं व्यासाष्टकं सम्पूर्णम्॥

\sect{श्रीकाञ्चीकामकोटिपीठजगद्गुरुपरम्परास्तवः}

\centerline{\small (पञ्चषष्टितमैः पीठाधिपतिभिः श्रीमत्सुदर्शनमहादेवेन्द्रसरस्वतीश्रीचरणैः प्रणीतः)}

\begingroup
\centering

\newcommand{\shvskip}{2ex}
\vspace{2ex}

[गुरवे सर्व-लोकानां भिषजे भव-रोगिणाम् ।\\
निधये सर्व-विद्यानां दक्षिणामूर्तये नमः ॥ *० ॥]\\[\shvskip]

नारायणं पद्मभुवं वसिष्ठं शक्तिं च तत्-पुत्र-पराशरं च ।\\
व्यासं शुकं गौडपदं महान्तं गोविन्द-योगी·न्द्र·मथा·स्य शिष्यम् ॥ १ ॥\\[\shvskip]

श्री-शङ्करा·चार्य·मथा·स्य पद्म-पादं च हस्तामलकं च शिष्यम् ।\\
तं तोटकं वार्तिक-कार·मन्यान् अस्मद्-गुरून् सन्तत·मानतोऽस्मि ॥ २ ॥\\[\shvskip]

सदाशिव-समारम्भां शङ्करा·चार्य-मध्यमाम् ।\\
अस्म·दाचार्य-पर्यन्तां वन्दे गुरु-परम्पराम् ॥ ३ ॥\\[\shvskip]

\textcolor{ShriGreen}{(१)} सर्व-तन्त्र-स्वतन्त्राय सदाऽऽत्मा·द्वैत-वेदिने ।\\
श्रीमते शङ्करा·र्याय वेदान्त-गुरवे नमः ॥ ४ ॥\\[\shvskip]

\textcolor{ShriGreen}{(*)} अविप्लुत-ब्रह्मचर्यान् अन्विते·न्द्र-सरस्वतीन् ।\\
आत्त-मिथ्यावार-पथान् अद्वैता·चार्य-सङ्कथान् ॥ ५ ॥\\[\shvskip]

आ-सेतु-हिमव·च्छैलं स·दाचार-प्रवर्तकान् ।\\
जगद्-गुरून् स्तुमः काञ्ची-शारदा-मठ-संश्रयान् ॥ ६ ॥\\[\shvskip]

\textcolor{ShriGreen}{(२)} पवित्रिते·तरा·द्वैत-मठ-पीठी-शिरो·भुवे ।\\
श्री-काञ्ची-शारदा-पीठ-गुरवे भव-भीरवे ॥ ७ ॥\\[\shvskip]

वार्तिका·दि-ब्रह्म-विद्या-कर्त्रे ब्रह्मा·वतारिणे ।\\
सुरेश्वरा·चार्य-नाम्ने योगी·न्द्राय नमो नमः ॥ ८ ॥\\[\shvskip]

\textcolor{ShriGreen}{(३)} अपोऽश्न·न्नेव जैनान् य आ-प्राग्ज्योतिष·माच्छिनत् ।\\
शिशु·माचार्य-वाग्-वेणी-रय-रोधि-महो·बलम् ॥ ९ ॥\\[\shvskip]

सङ्क्षेप-शारीर-मुख-प्रबन्ध-विवृता·द्वयम् ।\\
ब्रह्मस्वरूपा·र्य-भाष्य-शा·न्त्याचार्यक-पण्डितम् ॥ १० ॥\\[\shvskip]

सर्वज्ञ-चन्द्र-नाम्ना च सर्वतो भुवि विश्रुतम् ।\\
सर्वज्ञ-सद्-गुरुं वन्दे सर्वज्ञ·मिव भू-गतम् ॥ ११ ॥\\[\shvskip]

\textcolor{ShriGreen}{(४)} मेधाविनं सत्यबोधं व्याधूत-विमतो·च्चयम् ।\\
प्राच्य-भाष्य-त्रय-व्याख्या-प्रवीणं प्रभु·माश्रये ॥ १२ ॥\\[\shvskip]

\textcolor{ShriGreen}{(५)} ज्ञानानन्द-मुनी·न्द्रा·र्यं ज्ञानोत्तम-परा·भिधम् ।\\
चन्द्रचूड-पदा·सक्तं चन्द्रिका-कृत·माश्रये ॥ १३ ॥\\[\shvskip]

\textcolor{ShriGreen}{(६)} शुद्धानन्द-मुनी·न्द्राणां विद्धा·र्हत-मत-त्विषाम् ।\\
आनन्दज्ञान-सेव्यानाम् आलम्बे चरणा·म्बुजम् ॥ १४ ॥\\[\shvskip]

\textcolor{ShriGreen}{(७)} सर्व-शाङ्कर-भाष्यौ·घ-भाष्य-कर्तार·मद्वयम् ।\\
सर्व-वार्तिक-सद्-वृत्ति-कृतं श्रीशैल-गं भजे ॥ १५ ॥\\[\shvskip]

\textcolor{ShriGreen}{(८)} कैवल्यानन्द-योगी·न्द्रान् केवलं राज-योगिनः ।\\
कैवल्य-मात्र-निरतान् कलयेम जगद्-गुरून् ॥ १६ ॥\\[\shvskip]

\textcolor{ShriGreen}{(९)} श्री-कृपाशङ्करा·र्याणां मर्यादा·तीत-तेजसाम् ।\\
ष·ण्मता·चार्यक-जुषाम् अङ्घ्रि-द्वन्द्व·महं श्रये ॥ १७ ॥\\[\shvskip]

\textcolor{ShriGreen}{(१०)} महिष्ठाय नम·स्तस्मै महादेवाय योगिने ।\\
सुरेश्वरा·परा·ख्याय गुरवे दोष-भीरवे ॥ १८ ॥\\[\shvskip]

\textcolor{ShriGreen}{(११)} स्तुमः सदा शिवानन्द-चिद्घने·न्द्र-सरस्वतीन् ।\\
कामाक्षी-चन्द्रमौ·ल्यर्चा-कलनै·क-लस·न्मतीन् ॥ १९ ॥\\[\shvskip]

\textcolor{ShriGreen}{(१२)} सार्वभौमा·भिध-महा-व्रत-चर्या-परायणान् ।\\
वन्दे जगद्-गुरूं·श्चन्द्रशेखरे·न्द्र-सरस्वतीन् ॥ २० ॥\\[\shvskip]

\textcolor{ShriGreen}{(१३)} समा-द्वात्रिंश·दत्युग्र-काष्ठ-मौन-समाश्रयान् ।\\
जित-मृत्यून् महा-लिङ्ग-भूतान् सच्चिद्घनान् नुमः ॥ २१ ॥\\[\shvskip]

\textcolor{ShriGreen}{(१४)} महा-भैरव-दुस्तन्त्र-दुर्दान्त-ध्वान्त-भास्करान् ।\\
विद्याघनान् नमस्यामि सर्व-विद्या-विचक्षणान् ॥ २२ ॥\\[\shvskip]

\textcolor{ShriGreen}{(१५)} आचार्य-पद-पाथोज-परिचर्या-परायणम् ।\\
गङ्गाधरं नमस्यामः सदा गङ्गाधरा·र्चकम् ॥ २३ ॥\\[\shvskip]

\textcolor{ShriGreen}{(१६)} जग·ज्जयि-सु-सौराष्ट्र-जरदृष्टि-मदा·पहान् ।\\
शक-सिल्हक-दर्प-घ्नान् ईडीमहि महायतीन् ॥ २४ ॥\\[\shvskip]

\textcolor{ShriGreen}{(१७)} चतु·स्समुद्री-क्रोड-स्थ-वर्णा·श्रम-विचारकान् ।\\
श्रित-विप्र-व्रज-स्कन्ध-सुवर्णा·न्दोलिका-चरान् ॥ २५ ॥\\[\shvskip]

प्र·त्यहं ब्रह्म-साहस्र-सन्तर्पण-धृत-व्रतान् ।\\
सदाशिव-समाह्वानान् स्मरामः सद्-गुरून् सदा ॥ २६ ॥\\[\shvskip]

\textcolor{ShriGreen}{(१८)} माया-लोकायती-भूत-बृहस्पति-मदापहान् ।\\
वन्दे सुरेन्द्र-वन्द्या·ङ्घ्रीन् श्री-सुरेन्द्र-सरस्वतीन् ॥ २७ ॥\\[\shvskip]

\textcolor{ShriGreen}{(१९)} श्रीविद्या-करुणा-लब्ध-ब्रह्म-विद्या-हृता·मयान् ।\\
वन्दे वशंवद-प्राणान् मुनीन् विद्याघनान् मुहुः ॥ २८ ॥\\[\shvskip]

\textcolor{ShriGreen}{(२०)} विद्याघन-कृपा-लब्ध-सर्व-वेदान्त-विस्तरम् ।\\
कौतस्कुतो·त्पात-केतुं निश्शङ्कं नौमि शङ्करम् ॥ २९ ॥\\[\shvskip]

\textcolor{ShriGreen}{(२१)} चन्द्रचूड-पद-ध्यान-प्राप्ता·नन्द-महोदधीन् ।\\
यती·न्द्रां·श्चन्द्रचूडे·न्द्रान् स्मरामि मनसा सदा ॥ ३० ॥\\[\shvskip]

\textcolor{ShriGreen}{(२२)} नमामि परिपूर्ण-श्री-बोधान् ग्रावा·भिलापकान् ।\\
य·दीक्षणात् पलायन्त प्राणिना·मामया·धयः ॥ ३१ ॥\\[\shvskip]

\textcolor{ShriGreen}{(२३)} सच्चित्सुखान् प्रपद्येऽहं सुख·माप्त-गुहा-स्थितीन् ।\\
\textcolor{ShriGreen}{(२४)} चित्सुखा·चार्य·मीडेऽहं सत्सुखं कोङ्कणा·श्रयम् ॥\\[\shvskip]

\textcolor{ShriGreen}{(२५)} भजे श्री-सच्चिदानन्द-घने·न्द्रान् रस-साधनात् ।\\
लिङ्गा·त्मना परिणतान् प्रभासे योग-संश्रिते ॥ ३३ ॥\\[\shvskip]

\textcolor{ShriGreen}{(२६-२७-२८)} भगवत्पाद-पादा·ब्जा·सक्ति-निर्णिक्त-मानसान् ।\\
प्रज्ञाघनं चिद्विलासं महादेवं च मैथिलम् ॥ ३४ ॥\\[\shvskip]

\textcolor{ShriGreen}{(२९-३०)} पूर्णबोधं च बोधं च भक्ति-योग-प्रवर्तकम् ।\\
\textcolor{ShriGreen}{(३१)} ब्रह्मानन्दघने·न्द्रं च नमामि नियता·त्मनः ॥ ३५ ॥\\[\shvskip]

\textcolor{ShriGreen}{(३२)} चिदानन्दघने·न्द्राणां लम्बिका-योग-सेविनाम् ।\\
जीर्ण-पर्णा·शिनां पादौ प्रपद्ये मनसा सदा ॥ ३६ ॥\\[\shvskip]

\textcolor{ShriGreen}{(३३)} सच्चिदानन्द-नामानं शिवा·र्चन-परायणम् ।\\
भाषा-पञ्चदशी-प्राज्ञं भावयामि सदा मुदा ॥ ३७ ॥\\[\shvskip]

\textcolor{ShriGreen}{(३४)} भू-प्रदक्षिण-कर्मै·क-सक्तं श्री-चन्द्रशेखरम् ।\\
त्रात-दावा·ग्नि-सन्दग्ध-किशोरक·मुपास्महे ॥ ३८ ॥\\[\shvskip]

\textcolor{ShriGreen}{(३५)} चित्सुखे·न्द्रं सुखेनै·व क्रान्त-सह्य-गुहा-गृहम् ।\\
काम-रूप-चरं नाना-रूप-वन्त·मुपास्महे ॥ ३९ ॥\\[\shvskip]

\textcolor{ShriGreen}{(३६)} नि·र्दोष-संयम-धरान् चित्सुखानन्द-तापसान् ।\\
\textcolor{ShriGreen}{(३७)} विद्याघने·न्द्रान् श्रीविद्या-वशी-कृत-जनान् स्तुमः ॥\\[\shvskip]

\textcolor{ShriGreen}{(३८)} शङ्करे·न्द्र-यती·न्द्राणां पादुके ब्रह्म-सम्भृते ।\\
नमामि शिरसा याभ्यां त्रीन् लोकान् व्यचर·न्मुनिः ॥ ४१ ॥\\[\shvskip]

\textcolor{ShriGreen}{(३९-४०)} सच्चिद्विलास-योगी·न्द्रं महादेवे·न्द्र·मुज्ज्वलम् ।\\
\textcolor{ShriGreen}{(४१)} गङ्गाधरे·न्द्र·मप्येतान् नौमि वादि-शिरोमणीन् ॥\\[\shvskip]

\textcolor{ShriGreen}{(४२-४३)} ब्रह्मानन्दघने·न्द्रा·ख्यां·स्तथाऽऽनन्दघना·नपि ।\\
\textcolor{ShriGreen}{(४४)} पूर्णबोध-महर्षीं·श्च ज्ञान-निष्ठा·नुपास्महे ॥ ४३\\[\shvskip]

\textcolor{ShriGreen}{(४५)} वृत्त्याऽऽजगर्या श्रीशैल-गुहा-गृह-कृत-स्थितीन् ।\\
श्रीमत्-परशिवा·भिख्यान् सर्वा·तीतान् श्रये सदा ॥ ४४ ॥\\[\shvskip]

\textcolor{ShriGreen}{(४६-४७)} अन्योन्य-सदृशा·न्योन्यौ बोध-श्री-चन्द्रशेखरौ ।\\
प्रणवो·पासना-सक्त-मानसौ मनसा श्रये ॥ ४५ ॥\\[\shvskip]

\textcolor{ShriGreen}{(४८)} मुक्ति-लिङ्गा·र्चना·नन्द-विस्मृता·शेष-वृत्तये ।\\
चिदम्बर-रह·स्यन्त·र्लीन-देहाय योगिने ॥ ४६ ॥\\[\shvskip]

अद्वैता·नन्द-साम्राज्य-विद्रुता·शेष-पाप्मने ।\\
अद्वैतानन्दबोधाय नमो ब्रह्म समीयुषे ॥ ४७ ॥\\[\shvskip]

\textcolor{ShriGreen}{(४९-५०)} श्रये महादेव-चन्द्रशेखरे·न्द्र-महामुनी ।\\
महाव्रत-समारब्ध-कोटि-होमा·न्त-गामिनौ ॥ ४८ ॥\\[\shvskip]

\textcolor{ShriGreen}{(५१)} विद्यातीर्थ-समाह्वानान् श्रीविद्या-नाथ-योगिनः ।\\
विद्यया शङ्कर-प्रख्यान् विद्यारण्य-गुरून् भजे ॥ ४९ ॥\\[\shvskip]

\textcolor{ShriGreen}{(५२)} शङ्करानन्द-योगी·न्द्र-पद-पङ्कजयो·र्युगम् ।\\
बुक्क-भूप-शिरो·रत्नं स्मरामि सततं हृदा ॥ ५४ ॥\\[\shvskip]

\textcolor{ShriGreen}{(५३)} श्री-पूर्णानन्द-मौनी·न्द्रं नेपाल-नृप-देशिकम् ।\\
अव्याहत-स्व-सञ्चारं संश्रयामि जगद्-गुरुम् ॥ ५५ ॥\\[\shvskip]

\textcolor{ShriGreen}{(५४-५५)} महादेव·श्च त·च्छिष्य·श्चन्द्रशेखर-यो·ग्यपि ।\\
स्तां मे हृदि सदा धीरा·वद्वैत-मत-देशिकौ ॥ ५६ ॥\\[\shvskip]

\textcolor{ShriGreen}{(५६)} प्रवीर-सेतु-भूपाल-सेविता·ङ्घ्रि-सरोरुहान् ।\\
भजे सदाशिवे·न्द्र-श्री-बोधे·श्वर-गुरून् सदा ॥ ५७ ॥\\[\shvskip]

\textcolor{ShriGreen}{(५७)} सदाशिव-श्री-ब्रह्मेन्द्र-धृत-स्व-पद-पादुकान् ।\\
धीरान् परशिवे·न्द्रा·र्यान् ध्यायामि सततं हृदि ॥ ५८ ॥\\[\shvskip]

\textcolor{ShriGreen}{(५८)} आत्मबोध-यती·न्द्राणा·मा-शीता·चल-चारिणाम् ।\\
अन्य-श्री-शङ्करा·चार्य-धी-कृता·मङ्घ्रि·माश्रये ॥ ५९ ॥\\[\shvskip]

\textcolor{ShriGreen}{(५९)} भगवन्नाम-साम्राज्य-लक्ष्मी-सर्वस्व-विग्रहान् ।\\
श्रीमद्-बोधेन्द्र-योगी·न्द्र-देशिके·न्द्रा·नुपास्महे ॥ ६० ॥\\[\shvskip]

\textcolor{ShriGreen}{(६०)} अद्वैतात्मप्रकाशाय सर्व-शास्त्रा·र्थ-वेदिने ।\\
विधूत-सर्व-भेदाय नमो विश्वा·तिशायिने ॥ ६१ ॥\\[\shvskip]

\textcolor{ShriGreen}{(६१)} आ सप्तमा·ज्जीर्ण-पर्ण-जल-वाता·रुणां·शुभिः ।\\
कृत-स्व-प्राण-यात्राय महादेवाय सन्नतिः ॥ ६२ ॥\\[\shvskip]

\textcolor{ShriGreen}{(६२)} चोल-केरल-चेरौ·ड्र-पाण्ड्य-कर्णाट-कोङ्कणान् ।\\
महाराष्ट्रा·न्ध्र-सौराष्ट्र-मगधा·दीं·श्च भू-भुजः ॥ ६३ ॥\\[\shvskip]

शिष्या·ना-सेतु-शीता·द्रि शासते पुण्य-कर्मणे ।\\
श्री-चन्द्रशेखरे·न्द्राय जगतो गुरवे नमः ॥ ६४ ॥\\[\shvskip]

\textcolor{ShriGreen}{(६३)} निष्पाप-वृत्तये नित्य-निर्धूत-भव-कॢप्तये ।\\
महादेवाय सततं नमोऽस्तु नत-रक्षिणे ॥ ६५ ॥\\[\shvskip]

\textcolor{ShriGreen}{(६४)} श्रीविद्यो·पासना-दार्ढ्य-वशी-कृत-चराचरान् ।\\
श्री-चन्द्रशेखरे·न्द्रा·र्यान् शङ्कर-प्रतिमान् नुमः ॥ ६६ ॥\\[\shvskip]

॥ परिशिष्टम् ॥\\[\shvskip]

\textcolor{ShriGreen}{(६५)} कलाना·माश्रयं देवी-सान्निध्या·नुभुवं सदा ।\\
सुदर्शन-महादेव-गुरुं सत्ये·क्षणं नुमः ॥ *१ ॥\\[\shvskip]

\textcolor{ShriGreen}{(६६)} अद्वैत-रक्षणे विज्ञान् वाग्मी यः प्रैरयद् दृढम् ।\\
श्री-चन्द्रशेखरे·न्द्रो मे धुनो·त्वान्तर-कल्मषम् ॥ *२ ॥\\[\shvskip]

\textcolor{ShriGreen}{(६७)} गुरु-शुश्रूषणा·सक्ति-समर्पित-निजा·खिलम् ।\\
युवानं शान्ति-भूमानं महादेवं गुरुं श्रये ॥ *३ ॥\\[\shvskip]

\textcolor{ShriGreen}{(६८)} अपार-करुणा-सिन्धुं ज्ञान-दं शान्त-रूपिणम् ।\\
श्री-चन्द्रशेखर-गुरुं प्रणमामि मुदाऽन्वहम् ॥ *४ ॥\\[\shvskip]

\textcolor{ShriGreen}{(६९)} देवे देहे च देशे च भ·क्त्यारोग्य-सुख-प्रदम् ।\\
बुध-पामर-सेव्यं तं श्री-जयेन्द्रं नमा·म्यहम् ॥ *५ ॥\\[\shvskip]

\textcolor{ShriGreen}{(७०)} नमामः शङ्करा·न्वाख्य-विजये·न्द्र-सरस्वतीम् ।\\
श्री-गुरुं शिष्ट-मार्गा·नुनेतारं स·न्मति-प्रदम् ॥ *६ ॥\\[\shvskip]

(*) श्री-काञ्ची-शारदा-पीठ-संस्थिताना·मिमां क्रमात् ।\\
स्तुतिं जगद्-गुरूणां यः पठेत् स सुख-भाग् भवेत् ॥ ६७ ॥\\

\sect{व्यासाष्टोत्तरशतनामस्तोत्रम्}

\twolineshloka
{नारायणकुलोद्भूतो नारायणपरो वरः}
{नारायणावतारश्च नारायणवशंवदः}

\twolineshloka
{स्वयम्भूवंशसम्भूतो वसिष्ठकुलदीपकः}
{शक्तिपौत्रः पापहन्ता पराशरसुतोऽमलः}

\twolineshloka
{द्वैपायनो मातृभक्तः शिष्टः सत्यवतीसुतः}
{स्वयमुद्भूतवेदश्च चतुर्वेदविभागकृत्}

\twolineshloka
{महाभारतकर्ता च ब्रह्मसूत्रप्रजापतिः}
{अष्टादशपुराणानां कर्ता श्यामः प्रशिष्यकः}

\twolineshloka
{शुकतातः पिङ्गजटः प्रांशुर्दण्डी मृगाजिनः}
{वश्यवाग् ज्ञानदाता च शङ्करायुःप्रदः शुचिः}

\twolineshloka
{मातृवाक्यकरो धर्मी कर्मी तत्त्वार्थदर्शकः}
{सञ्जयज्ञानदाता च प्रतिस्मृत्युपदेशकः}

\twolineshloka
{सर्वधर्मोपदेष्टा च मृतदर्शनपण्डितः}
{विचक्षणः प्रहृष्टात्मा पर्वपूज्यः प्रभुर्मुनिः}

\twolineshloka
{वीरो विश्रुतविज्ञानः प्राज्ञश्चाज्ञाननाशनः}
{ब्राह्मकृत् पाद्मकृद् धीरो विष्णुकृच्छिवकृत् तथा}

\twolineshloka
{श्रीभागवतकर्ता च भविष्यरचनादरः}
{नारदाख्यस्य कर्ता च मार्कण्डेयकरोऽग्निकृत्}

\twolineshloka
{ब्रह्मवैवर्तकर्ता च लिङ्गकृच्च वराहकृत्}
{स्कान्दकर्ता वामनकृत् कूर्मकर्ता च मत्स्यकृत्}

\twolineshloka
{गरुडाख्यस्य कर्ता च ब्रह्माण्डाख्यपुराणकृत्}
{उपपुराणानां कर्ता पुराणः पुरुषोत्तमः}

\twolineshloka
{काशिवासी ब्रह्मनिधिर्गीतादाता महामतिः}
{सर्वज्ञः सर्वसिद्धिश्च सर्वशास्त्रप्रवर्तकः}

\twolineshloka
{सर्वाश्रयः सर्वहितः सर्वः सर्वगुणाश्रयः}
{विशुद्धः शुद्धिकृद् दक्षो विष्णुभक्तः शिवार्चकः}

\twolineshloka
{देवीभक्तः स्कन्दरुचिर्गणेशादृच्च योगवित्}
{पैलाचार्य ऋचः कर्ता शाकल्यार्यश्च याजुषः}

\twolineshloka
{जैमिन्यार्यः सामकर्ता सुमन्त्वार्योऽप्यथर्वकृत्}
{रोमहर्षणसूतार्यो लोकाचार्यो महामुनिः}

\twolineshloka
{व्यासकाशीरतिर्विश्वपूज्यो विश्वेशपूजकः}
{शान्तः शान्ताकृतिः शान्तचित्तः शान्तिप्रदस्तथा}

॥इति व्यासाष्टोत्तरशतनामस्तोत्रं सम्पूर्णम्॥
